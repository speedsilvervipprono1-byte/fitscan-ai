async function callGeminiVisionApi(base64Image, mimeType, imageDataUrl) { const libraryNames = FOOD_DATABASE.map(f => f.name).join('; '); // Gọi tới Serverless Function bí mật của Vercel const response = await fetch('/api/analyze', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ base64Image, mimeType, libraryNames }) }); if (!response.ok) { const errData = await response.json().catch(() => ({})); throw new Error(errData.error || `Lỗi server HTTP ${response.status}`); } const r = await response.json(); if (r.isFood === false) { const e = new Error('NOT_FOOD'); e.notFood = true; throw e; } const lib = r.libraryName ? FOOD_DATABASE.find(f => normalizeFoodLabel(f.name) === normalizeFoodLabel(r.libraryName)) : null; const confText = typeof r.confidence === 'number' ? ` · tin cậy ${Math.round(r.confidence * 100)}%` : ''; showRecognitionNotice('Gemini nhận diện', `${r.foodName || 'Món ăn'}${confText}`, lib ? `Khớp thư viện: ${lib.name}` : 'Số liệu dinh dưỡng do Gemini ước lượng'); if (lib) { displayScanResult({ ...lib, img: imageDataUrl }); } else { displayScanResult({ name: r.foodName || 'Món Quét AI', category: r.category || 'meat', calories: Math.round(r.calories) || 250, protein: r.protein ?? 20, carbs: r.carbs ?? 15, fat: r.fat ?? 10, serving: r.serving || '1 phần vừa', img: imageDataUrl, cheatAlert: r.cheatAlert ?? false, oilLevel: r.oilLevel || 'Thấp', cheatNote: r.cheatNote || 'Cân bằng dinh dưỡng tốt.', burnout: r.burnout || '25 phút vận động nhẹ.' }); } updateScanStatus('Gemini đã nhận diện ✓', false); }
